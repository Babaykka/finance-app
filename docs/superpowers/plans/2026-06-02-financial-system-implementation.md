```python
from __future__ import annotations

import argparse
from pathlib import Path

from openpyxl import Workbook
from openpyxl.formatting.rule import FormulaRule
from openpyxl.styles import Alignment, Font, PatternFill
from openpyxl.utils import get_column_letter
from openpyxl.worksheet.datavalidation import DataValidation

from finance_system.schema import LISTS, SHEETS, SheetSpec


DEFAULT_OUTPUT = Path("dist/finance-system-template.xlsx")
HEADER_FILL = PatternFill("solid", fgColor="1F2937")
HEADER_FONT = Font(color="FFFFFF", bold=True)
WARNING_FILL = PatternFill("solid", fgColor="FCE7F3")
CHECK_FILL = PatternFill("solid", fgColor="FEF3C7")


def build_workbook(output_path: Path = DEFAULT_OUTPUT) -> Path:
    workbook = Workbook()
    default = workbook.active
    workbook.remove(default)

    for sheet in SHEETS:
        worksheet = workbook.create_sheet(sheet.name)
        if sheet.name == "01 Главная":
            _setup_dashboard(worksheet)
        elif sheet.name == "06 Справочники":
            _setup_reference_sheet(worksheet)
        else:
            _setup_data_sheet(worksheet, sheet)

    _add_dropdowns(workbook)
    _add_conditional_formatting(workbook)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    workbook.save(output_path)
    return output_path


def _setup_data_sheet(worksheet, sheet: SheetSpec) -> None:
    worksheet.freeze_panes = "A2"
    for column_index, column in enumerate(sheet.columns, start=1):
        cell = worksheet.cell(row=1, column=column_index, value=column.title)
        cell.fill = HEADER_FILL
        cell.font = HEADER_FONT
        cell.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True)
        worksheet.column_dimensions[get_column_letter(column_index)].width = column.width
    worksheet.auto_filter.ref = worksheet.dimensions


def _setup_reference_sheet(worksheet) -> None:
    worksheet.freeze_panes = "A2"
    for column_index, (list_name, values) in enumerate(LISTS.items(), start=1):
        header = worksheet.cell(row=1, column=column_index, value=list_name)
        header.fill = HEADER_FILL
        header.font = HEADER_FONT
        header.alignment = Alignment(horizontal="center")
        worksheet.column_dimensions[get_column_letter(column_index)].width = max(20, min(42, len(list_name) + 6))
        for row_index, value in enumerate(values, start=2):
            worksheet.cell(row=row_index, column=column_index, value=value)


def _setup_dashboard(worksheet) -> None:
    worksheet["A1"] = "Финансовая система"
    worksheet["A1"].font = Font(size=18, bold=True)
    worksheet["A3"] = "Итоги по валютам и активам"
    worksheet["A3"].font = Font(bold=True)
    worksheet["A4"] = "Валюта/актив"
    worksheet["B4"] = "Места хранения"
    worksheet["C4"] = "Долги к получению"
    worksheet["D4"] = "План расходов"
    for cell in worksheet[4]:
        cell.fill = HEADER_FILL
        cell.font = HEADER_FONT

    for row_index, asset in enumerate(LISTS["ВалютыАктивы"], start=5):
        worksheet[f"A{row_index}"] = asset
        worksheet[f"B{row_index}"] = f"=SUMIF('02 Места хранения'!$F:$F,A{row_index},'02 Места хранения'!$G:$G)"
        worksheet[f"C{row_index}"] = f"=SUMIF('03 Долги'!$E:$E,A{row_index},'03 Долги'!$G:$G)"
        worksheet[f"D{row_index}"] = f"=SUMIF('07 Расходы-план'!$C:$C,A{row_index},'07 Расходы-план'!$D:$D)"

    worksheet["F3"] = "Ближайшие ожидаемые платежи"

        "('02 Места хранения'!C:C=\"у доверенного лица\")+"

def _add_dropdowns(workbook) -> None:
    list_positions = {name: index for index, name in enumerate(LISTS, start=1)}
    for sheet in SHEETS:
        if not sheet.columns:
            continue
        worksheet = workbook[sheet.name]
        for column_index, column in enumerate(sheet.columns, start=1):
            if not column.dropdown:
                continue
            list_column = list_positions[column.dropdown]
            list_letter = get_column_letter(list_column)
            values = LISTS[column.dropdown]
            formula = f"'06 Справочники'!${list_letter}$2:${list_letter}${len(values) + 1}"
            validation = DataValidation(type="list", formula1=formula, allow_blank=True)
            validation.error = "Выберите значение из справочника."
            validation.errorTitle = "Недопустимое значение"
            validation.prompt = "Используйте выпадающий список."
            validation.promptTitle = column.title
            worksheet.add_data_validation(validation)
            validation.add(f"{get_column_letter(column_index)}2:{get_column_letter(column_index)}1000")


def _add_conditional_formatting(workbook) -> None:
    storage = workbook["02 Места хранения"]
```

```javascript
const DASHBOARD_SHEET = '01 Главная';
const STORAGE_SHEET = '02 Места хранения';
const DEBTS_SHEET = '03 Долги';
const PAYMENTS_SHEET = '04 График и оплаты';

const STORAGE_TYPE_COLUMN = 3;
const STORAGE_ASSET_COLUMN = 6;
const STORAGE_AMOUNT_COLUMN = 7;
const STORAGE_CHECKED_AT_COLUMN = 16;
const STORAGE_STATUS_COLUMN = 17;
const STORAGE_PRIORITY_COLUMN = 18;

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('Финансы')
    .addItem('Отметить выбранные места проверенными', 'markSelectedStorageChecked')
    .addItem('Собрать блок внимания', 'refreshAttention')
    .addToUi();
}

function markSelectedStorageChecked() {
  const spreadsheet = SpreadsheetApp.getActive();
  const sheet = spreadsheet.getActiveSheet();
  if (sheet.getName() !== STORAGE_SHEET) {
    spreadsheet.toast('Откройте лист "02 Места хранения" и выберите строки.', 'Финансы', 5);
    return;
  }

  const range = sheet.getActiveRange();
  if (!range || range.getRow() < 2) {
    spreadsheet.toast('Выберите одну или несколько строк с местами хранения.', 'Финансы', 5);
    return;
  }

  const rowCount = range.getNumRows();
  const values = Array.from({ length: rowCount }, () => [new Date()]);
  sheet.getRange(range.getRow(), STORAGE_CHECKED_AT_COLUMN, rowCount, 1).setValues(values);
  spreadsheet.toast(`Обновлена дата проверки: ${rowCount} строк.`, 'Финансы', 5);
}

function refreshAttention() {
  const spreadsheet = SpreadsheetApp.getActive();
  const dashboard = spreadsheet.getSheetByName(DASHBOARD_SHEET);
  const storage = spreadsheet.getSheetByName(STORAGE_SHEET);

  if (!dashboard || !storage) {
    spreadsheet.toast('Не найдены листы Главная или Места хранения.', 'Финансы', 5);
    return;
  }

  const rows = storage.getDataRange().getValues().slice(1);
  const attentionRows = rows
    .filter(row => {
      const storageType = row[STORAGE_TYPE_COLUMN - 1];
      const status = row[STORAGE_STATUS_COLUMN - 1];
      const priority = row[STORAGE_PRIORITY_COLUMN - 1];
      return storageType === 'у доверенного лица' || status === 'проверить' || priority === 'срочно';
    })
    .map(row => [
      row[1],
      row[STORAGE_TYPE_COLUMN - 1],
      row[STORAGE_ASSET_COLUMN - 1],
      row[STORAGE_AMOUNT_COLUMN - 1],
      row[STORAGE_STATUS_COLUMN - 1],
      row[STORAGE_PRIORITY_COLUMN - 1],
    ]);

  dashboard.getRange('H4:M100').clearContent();
  dashboard.getRange('H4:M4').setValues([[
    'Место',
    'Тип',
    'Валюта/актив',
    'Сумма',
    'Статус',
    'Приоритет',
  ]]);

  if (attentionRows.length > 0) {
    dashboard.getRange(5, 8, attentionRows.length, 6).setValues(attentionRows);
  }

  spreadsheet.toast(`Блок внимания обновлен: ${attentionRows.length} строк.`, 'Финансы', 5);
}
```
